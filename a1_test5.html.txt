];
 
let vorname="";
let email="";
 
function starten(){
 
vorname=document.getElementById("vorname").value.trim();
email=document.getElementById("email").value.trim();
 
if(!vorname){
alert("Bitte Vorname eingeben.");
return;
}
 
if(!email){
alert("Bitte E-Mail eingeben.");
return;
}
 
if(!document.getElementById("zustimmung").checked){
alert("Bitte Zustimmung aktivieren.");
return;
}
 
document.getElementById("start").style.display="none";
document.getElementById("test").style.display="block";
 
fragenAufbauen();
}
 
function fragenAufbauen(){
 
let html="";
 
fragen.forEach((frage,index)=>{
 
html+=`
<div class="frage">
<h3>${index+1}. ${frage[0]}</h3>
`;
 
frage[1].forEach((antwort,aindex)=>{
 
html+=`
<label class="option">
<input type="radio" name="f${index}" value="${aindex}">
${antwort}
</label>
`;
 
});
 
html+=`</div>`;
 
});
 
document.getElementById("fragen").innerHTML=html;
 
}
 
async function auswerten(){
 
let punkte=0;
let offen=0;
 
fragen.forEach((f,index)=>{
 
const auswahl=document.querySelector(
'input[name="f'+index+'"]:checked'
);
 
if(!auswahl){
offen++;
return;
}
 
if(Number(auswahl.value)===f[2]){
punkte++;
}
 
});
 
if(offen>0){
alert("Bitte alle Fragen beantworten.");
return;
}
 
const prozent=Math.round(
(punkte/fragen.length)*100
);
 
const bestanden=prozent>=70;
 
document.getElementById("test").style.display="none";
document.getElementById("ergebnis").style.display="block";
 
document.getElementById("ergebnis").innerHTML=`
 
<div class="ergebnis">
 
<h2>🏆 Ergebnis A1 Test 5</h2>
 
<h3>${vorname}</h3>
 
<div class="level">
${prozent}%
</div>
 
<p>
${punkte} von ${fragen.length} Punkten
</p>
 
<div class="${
bestanden
?
"bestanden"
:
"nichtbestanden"
}">
${bestanden
?
"✅ Bestanden"
:
"❌ Nicht bestanden"}
</div>
 
<p>
Niveau:
<strong>A1</strong>
</p>
 
${
bestanden
?
"<p>🎉 Hervorragend! Du bist bereit für A2.</p>"
:
"<p>📚 Wiederhole noch einige A1-Themen und versuche es erneut.</p>"
}
 
<br><br>
 
<a href="index.html">
🏠 Zur Startseite
</a>
 
</div>
 
`;
 
sendenErgebnis(
punkte,
prozent,
bestanden
);
 
}
 
async function sendenErgebnis(
punkte,
prozent,
bestanden
){
 
try{
 
await fetch(
"https://api.web3forms.com/submit",
{
method:"POST",
headers:{
"Content-Type":"application/json",
"Accept":"application/json"
},
body:JSON.stringify({
 
access_key:ACCESS_KEY,
 
subject:
"A1 Abschlusstest 5 - " + vorname,
 
Vorname:vorname,
Email:email,
 
Test:"A1 Abschlusstest 5",
 
Punkte:
punkte + "/" + fragen.length,
 
Prozent:prozent,
 
Ergebnis:
bestanden
?
"Bestanden"
:
"Nicht bestanden"
 
})
}
);
 
}catch(error){
 
console.log(error);
 
}
 
}
