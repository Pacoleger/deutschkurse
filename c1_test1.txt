function starten(){
vorname=document.getElementById("vorname").value.trim();
email=document.getElementById("email").value.trim();
if(!vorname){alert("Bitte Vorname eingeben.");return;}
if(!email){alert("Bitte E-Mail eingeben.");return;}

document.getElementById("test").style.display="block";

function fragenAufbauen(){
let html="";
fragen.forEach((f,index)=>{
html+=`<div class="frage"><h3>${index+1}. ${f[0]}</h3>`;
f[1].forEach((antwort,aindex)=>{
html+=`<label class="option"><input type="radio" name="f${index}" value="${aindex}">${antwort}</label>`;
});
html+=`</div>`;
});
document.getElementById("fragen").innerHTML=html;
}
